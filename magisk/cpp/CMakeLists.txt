cmake_minimum_required(VERSION 3.18.1)
project(pikmin_control_core)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(A "${CMAKE_CURRENT_SOURCE_DIR}/../../native/automation/zygisk-module/cpp")
set(G "${CMAKE_CURRENT_SOURCE_DIR}/../../native/gps-copy/cpp")
get_filename_component(PCC_BUILD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O2 -fdata-sections -ffunction-sections -ffile-prefix-map=${PCC_BUILD_ROOT}=/pcc")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O2 -fno-exceptions -fno-rtti -fvisibility=hidden -ffile-prefix-map=${PCC_BUILD_ROOT}=/pcc")
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -ffixed-x18 -Wl,--hash-style=both -Wl,--gc-sections -Wl,--strip-all")
# One hook trampoline pool and one xdl implementation (GPS variant includes size_t casts).
file(GLOB XDL "${G}/xdl/*.c")
set(LSPLANT_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(vendor/lsplant EXCLUDE_FROM_ALL)
add_subdirectory(vendor/dobby EXCLUDE_FROM_ALL)
if(NOT EXISTS "${PCC_HC_DEX_DIR}/hc_dex.inc")
    message(FATAL_ERROR "Build the embedded HC DEX from tracked source first")
endif()
add_library(pikmin_control_core SHARED main.cpp hc_engine.cpp hc_compat.cpp "${A}/nectar_rpc.cpp" "${A}/seed_preview.cpp" "${G}/hack.cpp" "${G}/And64InlineHook.cpp" ${XDL})
target_include_directories(pikmin_control_core PRIVATE "${A}/support" "${G}/xdl/include")
target_link_libraries(pikmin_control_core log dl)
target_include_directories(pikmin_control_core PRIVATE "${PCC_HC_DEX_DIR}" vendor/dobby/include)
target_link_libraries(pikmin_control_core lsplant_static dobby_static)
# Bind embedded dependency calls locally, including when LSPosed remains for other modules.
target_link_options(pikmin_control_core PRIVATE "-Wl,--exclude-libs,ALL")
target_compile_definitions(pikmin_control_core PRIVATE PCC_UNIFIED=1 PCC_SEED_PREVIEW_GAME157=1)

if(PCC_HC_BUILD_PROBE)
    add_library(pcc_hc_engine_probe SHARED hc_engine.cpp ../tests/hc_engine_probe.cpp ${XDL})
    target_include_directories(pcc_hc_engine_probe PRIVATE . "${G}/xdl/include" vendor/dobby/include)
    target_link_libraries(pcc_hc_engine_probe lsplant_static dobby_static log dl)
    target_link_options(pcc_hc_engine_probe PRIVATE "-Wl,--exclude-libs,ALL")
endif()
