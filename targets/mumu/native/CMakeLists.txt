cmake_minimum_required(VERSION 3.18.1)
project(pcc_mumu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(PCC_REPO "" CACHE PATH "Canonical repository root")
set(PCC_AUTOMATION "" CACHE PATH "Selected MuMu automation sources")
if(NOT EXISTS "${PCC_REPO}/targets/mumu/source-set.json" OR NOT EXISTS "${PCC_AUTOMATION}/nectar_rpc.cpp" OR NOT EXISTS "${PCC_AUTOMATION}/seed_preview.cpp")
  message(FATAL_ERROR "Explicit MuMu source set required; no phone fallback")
endif()
set(B "${CMAKE_CURRENT_SOURCE_DIR}/bridge")
set(G "${PCC_REPO}/native/gps-copy/cpp")
if(ANDROID_ABI STREQUAL "x86_64")
  file(GLOB XDL_SOURCES "${G}/xdl/*.c")
  add_library(pcc_canary SHARED "${B}/loader.cpp" ${XDL_SOURCES})
  target_include_directories(pcc_canary PRIVATE "${G}/xdl/include" "${PCC_REPO}/native/automation/zygisk-module/cpp/support" "${PCC_REPO}/magisk/cpp")
  target_compile_definitions(pcc_canary PRIVATE PCC_CANARY_STAGE="entry" PCC_CANARY_GATE="manual" PCC_CANARY_NAMESPACE="created" PCC_CANARY_API="framework" PCC_CANARY_CORE=1)
elseif(ANDROID_ABI STREQUAL "arm64-v8a")
  file(GLOB XDL_SOURCES "${G}/xdl/*.c")
  add_library(pcc_canary SHARED "${B}/payload_core.cpp" "${PCC_AUTOMATION}/nectar_rpc.cpp" "${PCC_AUTOMATION}/seed_preview.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/gps/hack.cpp" "${G}/And64InlineHook.cpp" ${XDL_SOURCES})
  target_include_directories(pcc_canary PRIVATE "${B}" "${G}" "${G}/xdl/include" "${PCC_REPO}/native/automation/zygisk-module/cpp/support")
  target_compile_definitions(pcc_canary PRIVATE PCC_UNIFIED=1 PCC_SEED_PREVIEW_GAME157=1)
  target_link_options(pcc_canary PRIVATE -Wl,--strip-all -Wl,--no-undefined)
else()
  message(FATAL_ERROR "MuMu requires x86_64 loader plus ARM64 NativeBridge payload")
endif()
target_compile_options(pcc_canary PRIVATE -O2 -fno-exceptions -fno-rtti -fvisibility=hidden)
target_link_libraries(pcc_canary log dl)
